```ruby
class MagicLink < ApplicationRecord
  CODE_LENGTH = 6
  EXPIRATION = 15.minutes

  belongs_to :user

  scope :active, -> { where(expires_at: Time.current...) }

  before_validation :generate_code, :set_expiration, on: :create

  validates :code, presence: true, uniqueness: true

  def self.consume(code)
    active.find_by(code: code.to_s.strip)&.consume
  end

  def consume
    destroy
    self
  end

  private
    def generate_code
      self.code ||= loop do
        candidate = SecureRandom.random_number(10**CODE_LENGTH).to_s.rjust(CODE_LENGTH, "0")
        break candidate unless self.class.exists?(code: candidate)
      end
    end

    def set_expiration
      self.expires_at ||= EXPIRATION.from_now
    end
end
```

`app/models/challenge.rb`:

```ruby
class Challenge < ApplicationRecord
  TOKEN_LENGTH = 8

  belongs_to :user
  has_many :participants, dependent: :destroy

  before_validation :generate_token, on: :create

  validates :name, presence: { message: "dá um nome pro desafio" }
  validates :starts_on, presence: true
  validates :days, numericality: { only_integer: true, greater_than: 0, message: "precisa durar pelo menos 1 dia" }
  validates :token, presence: true, uniqueness: true

  scope :latest_first, -> { order(created_at: :desc) }

  def today = Date.current
  def ends_on = starts_on + days - 1
  def upcoming? = today < starts_on
  def finished? = today > ends_on
  def running? = !upcoming? && !finished?
  def date_for(day) = starts_on + day - 1

  def current_day
    return 0 if upcoming?
    [ (today - starts_on).to_i + 1, days ].min
  end

  private
    def generate_token
      self.token ||= loop do
        candidate = SecureRandom.base58(TOKEN_LENGTH)
        break candidate unless self.class.exists?(token: candidate)
      end
    end
end
```

```ruby
class Checkin < ApplicationRecord
  belongs_to :participant

  delegate :challenge, to: :participant

  validates :date, presence: true, uniqueness: { scope: :participant_id }
  validate :date_is_today_or_yesterday, on: :create
  validate :date_within_challenge, on: :create

  private
    def date_is_today_or_yesterday
      return if date.blank?
      unless [ challenge.today, challenge.today - 1 ].include?(date)
        errors.add :date, "só dá pra marcar hoje ou ontem"
      end
    end

    def date_within_challenge
      return if date.blank?
      unless (challenge.starts_on..challenge.ends_on).cover?(date)
        errors.add :date, "fora do período do desafio"
      end
    end
end
```

```ruby
module Authentication
  extend ActiveSupport::Concern

  included do
    before_action :resume_session
    helper_method :current_user, :signed_in?
  end

  private
    def resume_session
      Current.user = User.find_by(id: cookies.signed[:user_id])
    end

    def current_user = Current.user
    def signed_in? = current_user.present?

    def require_authentication
      unless signed_in?
        session[:return_to] = request.fullpath
        redirect_to new_session_path
      end
    end

    def sign_in(user)
      Current.user = user
      cookies.signed.permanent[:user_id] = { value: user.id, httponly: true, same_site: :lax }
    end

    def sign_out
      Current.user = nil
      cookies.delete(:user_id)
    end

    # Identidade de participante: um token permanente por navegador.
    def person_token
      cookies.signed[:person_token] || issue_person_token
    end

    def issue_person_token
      SecureRandom.base58(24).tap do |token|
        cookies.signed.permanent[:person_token] = { value: token, httponly: true, same_site: :lax }
      end
    end

    def current_participant_for(challenge)
```

```ruby
class ChallengesController < ApplicationController
  before_action :require_authentication, only: %i[ new create ]

  def index
    @active, @finished = my_challenges.partition { |challenge| !challenge.finished? }
  end

  def new
    @challenge = Challenge.new(starts_on: Date.tomorrow, days: 21)
  end

  def create
    @challenge = current_user.challenges.new(challenge_params)

    if @challenge.save
      @challenge.join name: params.expect(:participant_name), person_token: person_token, user: current_user
      redirect_to @challenge
    else
      render :new, status: :unprocessable_entity
    end
  end

  def show
    @challenge = Challenge.find(params[:id])
    @participant = current_participant_for(@challenge)

    redirect_to join_path(@challenge.token) if @participant.nil?
  end

  private
    def challenge_params
      params.expect(challenge: %i[ name starts_on days ])
    end

    def my_challenges
      Challenge.involving(person_token: cookies.signed[:person_token], user: current_user).latest_first
    end
end
```

`app/controllers/checkins_controller.rb`:

```ruby
class CheckinsController < ApplicationController
  def create
    challenge = Challenge.find(params.expect(:challenge_id))

    if participant = current_participant_for(challenge)
      participant.checkins.create(date: date_for(challenge))
      redirect_to challenge
    else
      head :forbidden
    end
  end

  def destroy
    checkin = Checkin.find(params[:id])

    if checkin.participant == current_participant_for(checkin.challenge)
      checkin.destroy
      redirect_to checkin.challenge
    else
      head :forbidden
    end
  end

  private
    def date_for(challenge)
      params[:date] == "yesterday" ? challenge.today - 1 : challenge.today
    end
end
```
